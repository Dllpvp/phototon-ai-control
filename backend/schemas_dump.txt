    """РЈР·РµР» РіСЂР°С„Р° РїСЂРёС‡РёРЅ РґР»СЏ API (В§9.1).

class GraphEdge(BaseModel):
    """Р РµР±СЂРѕ РіСЂР°С„Р° РїСЂРёС‡РёРЅ РґР»СЏ API (В§9.1).

    sign: Literal[-1, 1]
    weight: float
    curvature: float | None = None
    lag_days: int = 0
    note: str | None = None
